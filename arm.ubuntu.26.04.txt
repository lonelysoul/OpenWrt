*********************************************#!/bin/bash

set -e

echo "============================================================"
echo " OpenWrt 编译流程"
echo "============================================================"

# ============================================================
# 进入 OpenWrt 源码目录
# ============================================================

cd "$(dirname "$0")/openwrt"

echo
echo "============================================================"
echo " 更新 OpenWrt 主源码"
echo "============================================================"

git pull

# ============================================================
# 添加第三方 Feed
# ============================================================

echo
echo "============================================================"
echo " 配置第三方 Feeds"
echo "============================================================"

FEEDS_CONFIG="feeds.conf.default"
DDNSGO_FEED='src-git ddnsgo https://github.com/sirpdboy/luci-app-ddns-go.git'

if grep -qxF "$DDNSGO_FEED" "$FEEDS_CONFIG"; then
    echo "--> ddns-go Feed 已存在，跳过"
else
    echo "$DDNSGO_FEED" >> "$FEEDS_CONFIG"
    echo "--> 已添加 ddns-go Feed"
fi

# ============================================================
# Kernel 配置
# ============================================================

echo
echo "============================================================"
echo " 检查 Linux Kernel 配置"
echo "============================================================"

KERNEL_CONFIG="target/linux/x86/config-6.18"

if [ ! -f "$KERNEL_CONFIG" ]; then
    echo "❌ 未找到 Kernel 配置文件：$KERNEL_CONFIG"
    echo "   请确认当前 OpenWrt 使用的是 Linux 6.18。"
    exit 1
fi

echo "--> 检查：CONFIG_NETKIT=y"

if grep -qxF 'CONFIG_NETKIT=y' "$KERNEL_CONFIG"; then
    echo "  ✓ CONFIG_NETKIT=y 已存在，跳过"
else
    echo "CONFIG_NETKIT=y" >> "$KERNEL_CONFIG"
    echo "  + 已添加 CONFIG_NETKIT=y"
fi

echo "--> 检查：CONFIG_X86_NATIVE_CPU"

if grep -qE '^CONFIG_X86_NATIVE_CPU=|^# CONFIG_X86_NATIVE_CPU is not set$' "$KERNEL_CONFIG"; then
    echo "  ✓ CONFIG_X86_NATIVE_CPU 已存在，跳过"
else
    echo "# CONFIG_X86_NATIVE_CPU is not set" >> "$KERNEL_CONFIG"
    echo "  + 已添加 # CONFIG_X86_NATIVE_CPU is not set"
fi

echo "--> 检查：CONFIG_X86_SGX"

if grep -qE '^CONFIG_X86_SGX=|^# CONFIG_X86_SGX is not set$' "$KERNEL_CONFIG"; then
    echo "  ✓ CONFIG_X86_SGX 已存在，跳过"
else
    echo "# CONFIG_X86_SGX is not set" >> "$KERNEL_CONFIG"
    echo "  + 已添加 # CONFIG_X86_SGX is not set"
fi

echo
echo "--> 当前相关 Kernel 配置："

grep -E \
    '^CONFIG_NETKIT=|^CONFIG_X86_NATIVE_CPU=|^# CONFIG_X86_NATIVE_CPU is not set$|^CONFIG_X86_SGX=|^# CONFIG_X86_SGX is not set$' \
    "$KERNEL_CONFIG" || true

# ============================================================
# 更新并安装 Feeds
# ============================================================

echo
echo "============================================================"
echo " 更新与安装 Feeds"
echo "============================================================"

./scripts/feeds update -a
./scripts/feeds install -a

# ============================================================
# v2ray-geodata
# ============================================================

echo
echo "============================================================"
echo " 更新/拉取 v2ray-geodata"
echo "============================================================"

# 清理 feeds 中可能存在的同名 package
rm -rf feeds/packages/net/v2ray-geodata

# 使用指定版本的 v2ray-geodata package
rm -rf package/v2ray-geodata

git clone \
    https://github.com/lonelysoul/v2ray-geodata \
    package/v2ray-geodata

echo "--> v2ray-geodata 已准备完成"

# ============================================================
# KixDNS
# ============================================================

echo
echo "============================================================"
echo " 更新/拉取 KixDNS"
echo "============================================================"

KIXDNS_DIR="package/kixdns"

if [ -d "$KIXDNS_DIR/.git" ]; then

    echo "--> 检测到已存在 $KIXDNS_DIR，正在更新..."

    cd "$KIXDNS_DIR"

    # 放弃上一次脚本留下的本地修改，避免 git pull 冲突
    if ! git diff --quiet || ! git diff --cached --quiet; then
        echo "--> 检测到本地修改，正在恢复..."
        git reset --hard HEAD
    fi

    git pull

    cd ../..

else

    echo "--> 未检测到 $KIXDNS_DIR，正在克隆..."

    rm -rf "$KIXDNS_DIR"

    git clone \
        https://github.com/JohnsonRan/luci-app-kixdns.git \
        "$KIXDNS_DIR"

fi

echo "--> KixDNS 已准备完成"

# ============================================================
# luci-app-dae
# ============================================================

echo
echo "============================================================"
echo " 更新/拉取 luci-app-dae"
echo "============================================================"

DAE_DIR="package/dae"
DAE_MAKEFILE="$DAE_DIR/dae/Makefile"

if [ -d "$DAE_DIR/.git" ]; then

    echo "--> 检测到已存在 $DAE_DIR，正在更新..."

    cd "$DAE_DIR"

    # 放弃上一次脚本留下的本地修改，避免 git pull 冲突
    if ! git diff --quiet || ! git diff --cached --quiet; then
        echo "--> 检测到本地修改，正在恢复..."
        git reset --hard HEAD
    fi

    git pull

    cd ../..

else

    echo "--> 未检测到 $DAE_DIR，正在克隆..."

    rm -rf "$DAE_DIR"

    git clone \
        https://github.com/QiuSimons/luci-app-dae.git \
        "$DAE_DIR"

fi

# ============================================================
# 设置 DAE AMD64 v3 / AVX2
# ============================================================

echo
echo "============================================================"
echo " 设置 dae 为 AMD64 v3 / AVX2 编译"
echo "============================================================"

if [ ! -f "$DAE_MAKEFILE" ]; then
    echo "❌ 未找到 dae Makefile：$DAE_MAKEFILE"
    exit 1
fi

if grep -qE '^GO_AMD64[[:space:]]*:?=[[:space:]]*v3[[:space:]]*$' "$DAE_MAKEFILE"; then

    echo "--> GO_AMD64=v3 已存在，跳过修改"

else

    # 如果存在其他 GO_AMD64 设置，先删除
    sed -i '/^GO_AMD64[[:space:]]*:?=/d' "$DAE_MAKEFILE"

    # 放在 GO_PKG_BUILD_VARS 前面
    sed -i '/^GO_PKG_BUILD_VARS+=/i GO_AMD64:=v3' "$DAE_MAKEFILE"

    echo "--> 已设置 GO_AMD64:=v3"

fi

echo
echo "--> 当前 dae Go AMD64 配置："

grep -n -E '^GO_AMD64[[:space:]]*:?=' "$DAE_MAKEFILE" || true

# ============================================================
# 检查并处理 .config
# ============================================================

echo
echo "============================================================"
echo " 检查并处理配置文件 (.config)"
echo "============================================================"

if [ ! -f ".config" ]; then

    echo "--> 未检测到 .config 文件"
    echo "--> 启动 menuconfig 进行手动配置..."

    make menuconfig

else

    echo "--> 检测到已有 .config 文件"
    echo "--> 执行 defconfig 补全依赖..."

    make defconfig

fi

# ============================================================
# 下载源码
# ============================================================

echo
echo "============================================================"
echo " 下载源码"
echo "============================================================"

make download -j$(nproc) V=s

# ============================================================
# 开始编译
# ============================================================

echo
echo "============================================================"
echo " 开始编译"
echo "============================================================"

time make -j$(nproc) V=s

# ============================================================
# 编译完成
# ============================================================

echo
echo "============================================================"
echo " 编译完成"
echo "============================================================"

#!/bin/bash
set -e

OPENWRT_DIR="$HOME/openwrt"

cd "$OPENWRT_DIR"

echo "==> OpenWrt directory: $PWD"

cat > /tmp/openwrt-extra.config <<'EOF'
CONFIG_BIND_ENABLE_DOH=y
CONFIG_HTOP_LMSENSORS=y
CONFIG_LIBCURL_COOKIES=y
CONFIG_LIBCURL_FILE=y
CONFIG_LIBCURL_FTP=y
CONFIG_LIBCURL_HTTP=y
CONFIG_LIBCURL_HTTP2=y
CONFIG_LIBCURL_HTTP_AUTH=y
CONFIG_LIBCURL_MBEDTLS=y
CONFIG_LIBCURL_NO_SMB="!"
CONFIG_LIBCURL_PROXY=y
CONFIG_LIBCURL_UNIX_SOCKETS=y
CONFIG_LIBCURL_ZLIB=y
CONFIG_LIBNCURSES_TERMINFO=""
CONFIG_OPENSSL_ENGINE=y
CONFIG_OPENSSL_OPTIMIZE_SPEED=y
CONFIG_OPENSSL_WITH_ASM=y
CONFIG_OPENSSL_WITH_CHACHA_POLY1305=y
CONFIG_OPENSSL_WITH_CMS=y
CONFIG_OPENSSL_WITH_DEPRECATED=y
CONFIG_OPENSSL_WITH_ERROR_MESSAGES=y
CONFIG_OPENSSL_WITH_IDEA=y
CONFIG_OPENSSL_WITH_MDC2=y
CONFIG_OPENSSL_WITH_PSK=y
CONFIG_OPENSSL_WITH_SEED=y
CONFIG_OPENSSL_WITH_SRP=y
CONFIG_OPENSSL_WITH_TLS13=y
CONFIG_OPENSSL_WITH_WHIRLPOOL=y
CONFIG_PACKAGE_bind-dig=y
CONFIG_PACKAGE_bind-libs=y
CONFIG_PACKAGE_curl=y
CONFIG_PACKAGE_ethtool-full=y
CONFIG_PACKAGE_htop=y
CONFIG_PACKAGE_iftop=y
CONFIG_PACKAGE_iperf3=y
CONFIG_PACKAGE_kmod-netlink-diag=y
CONFIG_PACKAGE_libatomic=y
CONFIG_PACKAGE_libbpf=y
CONFIG_PACKAGE_libcurl=y
CONFIG_PACKAGE_libelf=y
CONFIG_PACKAGE_libiperf3=y
CONFIG_PACKAGE_libncurses=y
CONFIG_PACKAGE_libnghttp2=y
CONFIG_PACKAGE_libopenssl=y
CONFIG_PACKAGE_libpcap=y
CONFIG_PACKAGE_libpcre2=y
CONFIG_PACKAGE_libstdcpp=y
CONFIG_PACKAGE_liburcu=y
CONFIG_PACKAGE_libuv=y
CONFIG_PACKAGE_lsof=y
CONFIG_PACKAGE_mtr-json=y
CONFIG_PACKAGE_mtr-nojson=m
CONFIG_PACKAGE_netcat=y
CONFIG_PACKAGE_nmap=y
CONFIG_PACKAGE_socat=y
CONFIG_PACKAGE_ss=y
CONFIG_PACKAGE_strace=y
CONFIG_PACKAGE_tcpdump=y
CONFIG_PACKAGE_terminfo=y
CONFIG_PCRE2_JIT_ENABLED=y
EOF

echo
echo "==> Updating .config ..."

# 删除已有的同名配置
while IFS='=' read -r key value; do
    [ -z "$key" ] && continue
    sed -i "/^${key}=.*/d" .config
done < /tmp/openwrt-extra.config

# 追加新的配置
cat /tmp/openwrt-extra.config >> .config

echo "==> Running make defconfig ..."
make defconfig

echo
echo "==> Checking final configuration ..."
echo

while IFS='=' read -r key value; do
    [ -z "$key" ] && continue

    result=$(grep -E "^${key}=" .config || true)

    if [ -n "$result" ]; then
        echo "[OK]   $result"
    else
        echo "[MISS] $key"
    fi
done < /tmp/openwrt-extra.config

echo
echo "==> Done."
